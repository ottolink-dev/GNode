264 template <typename T> T *get_value_ref(const std::string &port_label) const
265 {
266 for (const auto &port : this->ports)
267 {
268 if (port->get_label() == port_label)
269 {
270 if (port->get_port_type() == PortType::IN)
271 {
272 auto inputPort = std::dynamic_pointer_cast<Input<T>>(port);
273 if (!inputPort)
274 throw std::runtime_error("Type mismatch for port: " + port_label);
275 return inputPort->get_value_ref();
276 }
277 else
278 {
279 auto outputPort = std::dynamic_pointer_cast<Output<T>>(port);
280 if (!outputPort)
281 throw std::runtime_error("Type mismatch for port: " + port_label);
282 return outputPort->get_value_ref();
283 }
284 }
285 }
286 throw std::runtime_error("Port not found: " + port_label);
287 }

296 template <typename T> T *get_value_ref(int port_index) const
