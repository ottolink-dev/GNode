235 template <typename T> T *get_value_ref(const std::string &port_label) const
236 {
237 for (const auto &port : this->ports)
238 {
239 if (port->get_label() == port_label)
240 {
241 if (port->get_port_type() == PortType::IN)
242 {
243 auto inputPort = std::dynamic_pointer_cast<Input<T>>(port);
244 if (!inputPort)
245 throw std::runtime_error("Type mismatch for port: " + port_label);
246 return inputPort->get_value_ref();
247 }
248 else
249 {
250 auto outputPort = std::dynamic_pointer_cast<Output<T>>(port);
251 if (!outputPort)
252 throw std::runtime_error("Type mismatch for port: " + port_label);
253 return outputPort->get_value_ref();
254 }
255 }
256 }
257 throw std::runtime_error("Port not found: " + port_label);
258 }
