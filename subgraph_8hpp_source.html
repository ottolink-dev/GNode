68template <typename T> class NodeIn : public Node, public ISubGraphInputNode
69{
70public:
71 NodeIn(const std::string &slot_name) : Node("NodeIn"), slot_name(slot_name)
72 {
73 this->add_port<T>(PortType::OUT, "value");
74 }
75
76 void compute() override {}
77
78 std::string get_slot_name() const override { return this->slot_name; }
79
80 void pull_from_parent(Node *parent_node,
81 const std::string &port_label) override
82 {
83 T *parent_val = parent_node->get_value_ref<T>(port_label);
84 T *inner_val = this->get_value_ref<T>("value");
85
86 if (parent_val && inner_val)
87 {
88 if (*inner_val != *parent_val)
89 {
90 *inner_val = *parent_val;
91 this->is_dirty = true;
92 }
93 }
94 }
95
96private:
97 std::string slot_name;
98};

118

120 {
121 T *inner_val = this->get_value_ref<T>("value");
122 T *parent_val = parent_node->get_value_ref<T>(port_label);
123
124 if (inner_val && parent_val)
125 {
126 *parent_val = *inner_val;
