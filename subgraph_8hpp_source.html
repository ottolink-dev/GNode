74template <typename T> class NodeIn : public Node, public ISubGraphInputNode
75{
76public:
77 NodeIn(const std::string &slot_name) : Node("NodeIn"), slot_name(slot_name)
78 {
79 this->add_port<T>(PortType::OUT, "value");
80 }
81
82 void compute() override {}
83
84 std::string get_slot_name() const override { return this->slot_name; }
85
86 void register_on_parent(Node *parent_node,
87 const std::string &port_label) const override
88 {
89 parent_node->add_port<T>(PortType::IN, port_label);
90 }
91
92 void pull_from_parent(Node *parent_node,
93 const std::string &port_label) override
94 {
95 T *parent_val = parent_node->get_value_ref<T>(port_label);
96 T *inner_val = this->get_value_ref<T>("value");
97
98 if (parent_val && inner_val)
99 {
100 if (*inner_val != *parent_val)
101 {
102 *inner_val = *parent_val;
103 this->is_dirty = true;
104 }
105 }
106 }
107
108private:
109 std::string slot_name;
110};
