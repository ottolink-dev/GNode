119 EventConnection subscribe(callback_t callback)
120 {
121 std::lock_guard<std::recursive_mutex> lock(*mutex_);
122
123 const listener_id_t id = next_id_++;
124
125 listeners_.push_back({id, std::move(callback)});
126
127 // Capture a weak handle to the Event's lifetime token. If the Event is
128 // destroyed before the connection disconnects, the token is gone and the
129 // disconnect becomes a safe no-op (prevents UAF on listeners_).
130 std::weak_ptr<void> alive = alive_token_;
131
132 return EventConnection(
133 [this, id, alive]()
134 {
135 if (!alive.expired()) this->unsubscribe(id);
136 });
137 }

145 void notify(Args... args)
146 {
147 // held while the callbacks run, recursive so that they can subscribe or
148 // disconnect themselves
149 std::lock_guard<std::recursive_mutex> lock(*mutex_);
150
151 // copy to allow callbacks to disconnect themselves safely
152 auto listeners = listeners_;
153
154 for (auto &listener : listeners)
155 {
156 listener.callback(args...);
157 }
158 }

163 listener_id_t id;

167 void unsubscribe(listener_id_t id)
168 {
169 std::lock_guard<std::recursive_mutex> lock(*mutex_);
170
171 listeners_.erase(std::remove_if(listeners_.begin(),
172 listeners_.end(),
173 [id](const Listener &listener)
174 { return listener.id == id; }),
175 listeners_.end());
176 }

178 std::vector<Listener> listeners_;

181
